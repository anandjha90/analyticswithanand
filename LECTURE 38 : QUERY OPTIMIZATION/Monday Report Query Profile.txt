-- BASELINE QUERY - Regional Sales Dashboard (SLOW VERSION)
SELECT 
    R.R_NAME AS REGION,
    DAYOFWEEK(O.O_ORDERDATE) AS DAY_ID,
    DAYNAME(O.O_ORDERDATE) AS WEEK_DAY_NAME,
    SUM(O.O_TOTALPRICE) AS TOTAL_SALES
FROM 
    ORDERS AS O 
    LEFT JOIN CUSTOMER AS C ON C.C_CUSTKEY = O.O_CUSTKEY
    LEFT JOIN NATION AS N ON N.N_NATIONKEY = C.C_NATIONKEY
    LEFT JOIN REGION AS R ON R.R_REGIONKEY = N.N_REGIONKEY
WHERE YEAR(O.O_ORDERDATE) = 1995
GROUP BY R.R_NAME, DAYOFWEEK(O.O_ORDERDATE), DAYNAME(O.O_ORDERDATE)
ORDER BY R.R_NAME, DAYOFWEEK(O.O_ORDERDATE);

-- Save this for analysis
SET baseline_query_id = LAST_QUERY_ID();
